C) Create `src/services/audioCaptureService.ts`:
A class-based service that wraps expo-audio's imperative API:

```typescript
import { createAudioRecorder, requestRecordingPermissionsAsync, setAudioModeAsync } from 'expo-audio';
import { WHISPER_QUALITY } from '@/utils/recordingPresets';
import * as FileSystem from 'expo-file-system';

export const RECORDING_STATES = {
  IDLE: 'idle',
  RECORDING: 'recording',
  PROCESSING: 'processing',
  CALL_PAUSED: 'call-paused',
  ERROR: 'error',
} as const;

export type RecordingStatus = (typeof RECORDING_STATES)[keyof typeof RECORDING_STATES];

export class AudioCaptureService {
  private recorder: ReturnType<typeof createAudioRecorder> | null = null;
  private tempFilePath: string | null = null;
  private meteringCallback: ((value: number) => void) | null = null;

  onMetering(cb: (value: number) => void) { this.meteringCallback = cb; }

  async requestPermissions(): Promise<boolean> {
    try {
      const { granted } = await requestRecordingPermissionsAsync();
      if (!granted) return false;
      await setAudioModeAsync({
        allowsRecording: true,
        playsInSilentMode: true,
        interruptionMode: 'mixWithOthers', // D-11: suppress notification interrupts
      });
      return true;
    } catch { return false; }
  }

  async startRecording(): Promise<boolean> {
    try {
      const granted = await this.requestPermissions();
      if (!granted) return false;

      const { createAudioRecorder: createRec } = await import('expo-audio');
      this.recorder = createRec(WHISPER_QUALITY);
      await this.recorder.prepareToRecordAsync();
      this.recorder.record();
      this.tempFilePath = this.recorder.uri;

      // Poll metering from recorder state
      // expo-audio exposes metering via recorder status listener
      this.pollMetering();

      return true;
    } catch {
      this.cleanupAfterError();
      return false;
    }
  }

  private async pollMetering() {
    // Metering polling loop — runs while recorder is active
    // Uses setInterval as fallback if event-based API not available
    const pollId = setInterval(() => {
      if (!this.recorder) { clearInterval(pollId); return; }
      try {
        // Access metering via internal status polling
        // expo-audio metering values may be accessed through the recorder's status
        // If metering is on a callback/event, use that instead
        const metering = (this.recorder as unknown as Record<string, unknown>).metering as number | undefined;
        if (metering !== undefined && metering !== null && this.meteringCallback) {
          const linear = metering <= -160 ? 0 : Math.pow(10, metering / 20);
          this.meteringCallback(Math.min(1, Math.max(0, linear)));
        }
      } catch { /* polling continues silently */ }
    }, 16); // ~60fps
  }

  async stopRecording(): Promise<string | null> {
    try {
      await this.recorder?.stop();
      const uri = this.recorder?.uri ?? null;
      this.recorder = null;
      return uri;
    } catch { return null; }
  }

  async cleanupTempFile(uri: string): Promise<void> {
    try {
      const fileInfo = await FileSystem.getInfoAsync(uri);
      if (fileInfo.exists) {
        await FileSystem.deleteAsync(uri, { idempotent: true });
      }
    } catch { /* best-effort cleanup */ }
  }

  private async cleanupAfterError() {
    this.recorder = null;
    if (this.tempFilePath) {
      await this.cleanupTempFile(this.tempFilePath);
      this.tempFilePath = null;
    }
  }

  getTempFilePath(): string | null { return this.tempFilePath; }
}

export const audioCaptureService = new AudioCaptureService();
```

Key design points:
- Singleton exported `audioCaptureService` used by hooks (not instantiated per-component)
- `requestPermissions` is called inside `startRecording` — permissions checked each time for user-revoked edge cases
- Metering polling at ~60fps with dBFS-to-linear conversion: `Math.pow(10, metering / 20)` (per Research Open Question 2)
- `interruptionMode: 'mixWithOthers'` suppresses notification audio interrupts (per D-11)
- All errors caught with best-effort cleanup

If `createAudioRecorder` imperative API does NOT exist in expo-audio v55 (per Research Assumption A2), use this fallback: create a thin wrapper that uses the `useAudioRecorder` hook at the component level and passes recorder control through props/context instead. Add a comment marking this fallback path.

If `expo-file-system` is not in package.json (it's not currently), the FileSystem-dependent cleanup will fail silently. The component-level callers catch errors — this is acceptable for Phase 1. Plan 01-02 installs expo-file-system explicitly.

A) Create `src/services/interruptionService.ts`:
A custom hook that listens to AppState changes for call detection and configures audio mode for notification suppression:

```typescript
import { useEffect, useRef } from 'react';
import { AppState, type AppStateStatus } from 'react-native';
import { setAudioModeAsync } from 'expo-audio';
import { useRecordingStore } from '@/stores/recordingStore';
import * as Haptics from 'expo-haptics';

export function useInterruptionHandler() {
  const appState = useRef(AppState.currentState);
  const { isRecording, setPaused, resume, addInterruptionMarker } = useRecordingStore();

  useEffect(() => {
    // Configure audio session to suppress notification interrupts (D-11)
    setAudioModeAsync({
      allowsRecording: true,
      playsInSilentMode: true,
      interruptionMode: 'mixWithOthers', // Don't pause for notifications/alarms
    });

    const subscription = AppState.addEventListener('change', (nextState: AppStateStatus) => {
      // Phone call detection (D-10):
      // iOS: call → inactive, call ends → active
      // Android: call → background, call ends → active
      if (appState.current === 'active' && nextState === 'inactive') {
        // Incoming call while recording
        if (isRecording) {
          setPaused(true);
          // Rive freezes, timer pauses, waveform dims (handled by store state)
        }
      } else if (appState.current === 'inactive' && nextState === 'active') {
        // Call ended — resume if was paused
        if (isRecording) {
          resume();
        }
      }
      appState.current = nextState;
    });

    // Notification/alarm interruption marker (D-11):
    // When the audio session is interrupted by a notification, expo-audio
    // fires an interruption event. Since we use mixWithOthers, recording
    // continues but we add a haptic buzz marker.
    // The marker is injected into the waveform via addInterruptionMarker.
    // For Phase 1, this is a stub that fires a light haptic on any
    // unexpected metering spike that could indicate an interruption.

    return () => subscription.remove();
  }, [isRecording, setPaused, resume, addInterruptionMarker]);
}
```

B) Create `src/stores/recordingStore.ts`:
A Zustand store slice managing recording state, metering, duration, and status:

```typescript
import { create } from 'zustand';

export type RecordingStatus = 'idle' | 'recording' | 'processing' | 'call-paused' | 'error';

interface RecordingState {
  isRecording: boolean;
  isPaused: boolean;
  isProcessing: boolean;
  duration: number;       // milliseconds
  metering: number;       // normalized 0-1 amplitude
  status: RecordingStatus;
  errorMessage: string | null;
  interruptionMarkers: number[]; // timestamps (in ms) where interruptions occurred

  // Actions
  setRecording: (val: boolean) => void;
  setPaused: (val: boolean) => void;
  setProcessing: (val: boolean) => void;
  setDuration: (ms: number) => void;
  setMetering: (val: number) => void;
  setStatus: (s: RecordingStatus) => void;
  setError: (msg: string | null) => void;
  resume: () => void;
  addInterruptionMarker: (timestampMs: number) => void;
  reset: () => void;
}

const initialState = {
  isRecording: false,
  isPaused: false,
  isProcessing: false,
  duration: 0,
  metering: 0,
  status: 'idle' as RecordingStatus,
  errorMessage: null,
  interruptionMarkers: [],
};

export const useRecordingStore = create<RecordingState>((set, get) => ({
  ...initialState,

  setRecording: (val) => set({ isRecording: val, status: val ? 'recording' : 'idle' }),

  setPaused: (val) => set({
    isPaused: val,
    status: val ? 'call-paused' : 'recording',
  }),

  setProcessing: (val) => set({
    isProcessing: val,
    status: val ? 'processing' : 'idle',
  }),

  setDuration: (ms) => set({ duration: ms }),

  setMetering: (val) => set({ metering: val }),

  setStatus: (s) => set({ status: s }),

  setError: (msg) => set({
    errorMessage: msg,
    status: msg ? 'error' : get().status === 'error' ? 'idle' : get().status,
  }),

  resume: () => set({
    isPaused: false,
    status: 'recording',
  }),

  addInterruptionMarker: (timestampMs) => set((state) => ({
    interruptionMarkers: [...state.interruptionMarkers, timestampMs],
  })),

  reset: () => set(initialState),
}));
```

Key design decisions:
- `status` field uses string union type for strict state machine (idle | recording | processing | call-paused | error)
- `metering` is normalized 0-1 for direct use by Skia waveform (Plan 01-02)
- `interruptionMarkers` array stores millisecond timestamps where haptic buzz markers appear on waveform
- `resume()` restores recording state after call ends (D-10)
- All state transitions in one store prevents stale state between multiple stores

Do NOT use `persist` middleware — this store is ephemeral per session. State is reset on app restart (Phase 2 adds Realm persistence).

```tsx
import { useEffect } from 'react';
import Animated, {
  useSharedValue,
  useAnimatedStyle,
  withTiming,
  withRepeat,
  withSequence,
  Easing,
} from 'react-native-reanimated';
import { View } from 'react-native';

interface GlowRingProps {
  isActive: boolean;
}

export function GlowRing({ isActive }: GlowRingProps) {
  const opacity = useSharedValue(0);
  const scale = useSharedValue(0.95);

  useEffect(() => {
    if (isActive) {
      // Pulse between 0.3 and 1.0 with 1.6s cycle (UI-SPEC §7.1, §9.1)
      opacity.value = withRepeat(
        withSequence(
          withTiming(0.3, { duration: 800, easing: Easing.inOut(Easing.sin) }),
          withTiming(1.0, { duration: 800, easing: Easing.inOut(Easing.sin) })
        ),
        -1,  // infinite
        true // reverse
      );
      scale.value = withTiming(1.0, { duration: 200 });
    } else {
      opacity.value = withTiming(0, { duration: 300 });
      scale.value = withTiming(0.95, { duration: 300 });
    }
  }, [isActive, opacity, scale]);

  const animatedStyle = useAnimatedStyle(() => ({
    opacity: opacity.value,
    transform: [{ scale: scale.value }],
  }));

  return (
    <View className="absolute w-[180px] h-[180px] items-center justify-center">
      <Animated.View
        className="w-full h-full rounded-full bg-primary/25"
        style={animatedStyle}
      />
    </View>
  );
}
```

The GlowRing is positioned behind the RecorderButton using absolute positioning (parent View handles the stack). It's 180×180px, 36px larger than the 144×144 button to create a visible glow halo.

```tsx
export function RecorderButton({ onStartRecording, onStopRecording }: RecorderButtonProps) {
  const riveRef = useRef<RiveRef>(null);
  const scale = useSharedValue(1);
  const { isRecording, isProcessing, status } = useRecordingStore();

  const buttonStyle = useAnimatedStyle(() => ({
    transform: [{ scale: scale.value }],
  }));

  const handlePress = useCallback(async () => {
    if (isProcessing) return; // Busy guard (UI-SPEC §7.2)

    if (!isRecording) {
      // START RECORDING (D-04, D-05)
      // Haptic FIRST, then spring (per Research Pitfall 4 — haptic ~10ms, spring first frame ~16ms)
      await Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium);

      // Spring bounce: 1.0 → 1.15 → settle at 1.0 (D-06)
      scale.value = withSpring(1.15, {
        mass: 0.5,
        stiffness: 200,
        damping: 12,
      }, () => {
        scale.value = withSpring(1.0);
      });

      // Rive morph to equalizer (D-15)
      try {
        riveRef.current?.setInputState('MicStateMachine', 'isRecording', true);
      } catch { /* Rive may not be loaded — component will show SVG fallback */ }

      onStartRecording();
    } else {
      // STOP RECORDING (D-04, D-05)
      await Haptics.notificationAsync(Haptics.NotificationFeedbackType.Success);

      // Brief pulse on stop
      scale.value = withSpring(1.05, {
        mass: 0.5,
        stiffness: 150,
        damping: 15,
      }, () => {
        scale.value = withSpring(1.0);
      });

      // Rive morph back to idle (D-15 reverse)
      try {
        riveRef.current?.setInputState('MicStateMachine', 'isRecording', false);
      } catch { /* fallback */ }

      onStopRecording();
    }
  }, [isRecording, isProcessing, onStartRecording, onStopRecording, scale]);

  return (
    <Pressable
      onPress={handlePress}
      disabled={isProcessing}
      accessibilityLabel={isRecording ? 'Stop recording' : 'Record audio'}
      accessibilityHint={
        isRecording
          ? 'Double tap to stop recording'
          : 'Double tap to start recording a voice entry'
      }
      className="w-[144px] h-[144px] items-center justify-center"
    >
      <Animated.View style={buttonStyle} className="w-[144px] h-[144px]">
        {/* Rive component — shows .riv animation when file is available */}
        <Rive
          ref={riveRef}
          resourceName="mic-to-equalizer"
          stateMachineName="MicStateMachine"
          style={{ width: 144, height: 144 }}
          autoplay={true}
        />
      </Animated.View>
      {/* SVG fallback if .riv not available — rendered behind Rive */}
      <View className="absolute" pointerEvents="none">
        <SvgXml
          xml={isRecording ? RECORDING_ICON : IDLE_ICON}
          width={72}
          height={72}
          color="#FF6B9E"
        />
      </View>
    </Pressable>
  );
}

// Fallback SVG icons for when .riv file is not available
const IDLE_ICON = `<svg viewBox="0 0 24 24" fill="none"><path d="M12 2a3 3 0 0 0-3 3v6a3 3 0 0 0 6 0V5a3 3 0 0 0-3-3z" fill="currentColor" opacity="0.6"/><path d="M6 11a6 6 0 0 0 12 0h2a8 8 0 0 1-7 7.93V21h-2v-2.07A8 8 0 0 1 4 11h2z" fill="currentColor"/></svg>`;

const RECORDING_ICON = `<svg viewBox="0 0 24 24" fill="none"><rect x="9" y="2" width="2" height="20" rx="1" fill="currentColor"/><rect x="13" y="4" width="2" height="16" rx="1" fill="currentColor" opacity="0.7"/><rect x="5" y="6" width="2" height="12" rx="1" fill="currentColor" opacity="0.4"/><rect x="17" y="8" width="2" height="8" rx="1" fill="currentColor" opacity="0.4"/></svg>`;
```

Key design decisions:
- **Haptic FIRST** before spring animation (per Research Pitfall 4): `Haptics.impactAsync(MEDIUM)` then immediately `scale.value = withSpring(1.15)`. The haptic registers ~10ms, spring first frame ~16ms = natural "impact → visual bounce" sequence.
- **Busy guard**: `if (isProcessing) return` — no taps during processing state
- **Rive fallback**: SVG icons shown behind the Rive canvas. If the .riv file isn't bundled, Rive renders blank and the SVG is visible through. This means the component works without the animation file.
- `accessibilityLabel`/`accessibilityHint` toggle based on recording state (UI-SPEC §7.2, §11)
- The `onStartRecording` / `onStopRecording` props are callbacks wired in HomeScreen to the audio capture service (Plan 01-02 adds the useAudioCapture hook; Plan 01-01 passes simple store-toggling callbacks)

**C) `src/components/ui/RecordingTimer.tsx`** (UI-SPEC §7.4):
```tsx
import { useEffect, useRef, useCallback } from 'react';
import { Text, View } from 'react-native';
import Animated, {
  useSharedValue,
  useAnimatedStyle,
  withTiming,
} from 'react-native-reanimated';
import { useRecordingStore } from '@/stores/recordingStore';

export function RecordingTimer() {
  const { isRecording, isPaused, duration } = useRecordingStore();
  const opacity = useSharedValue(0);
  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);

  // Format ms → mm:ss
  const formatTime = useCallback((ms: number): string => {
    const totalSeconds = Math.floor(ms / 1000);
    const minutes = Math.floor(totalSeconds / 60);
    const seconds = totalSeconds % 60;
    return `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
  }, []);

  // Timer increment every 100ms during recording (UI-SPEC §7.4 — update via setInterval)
  useEffect(() => {
    if (isRecording && !isPaused) {
      const startTime = Date.now() - duration;
      intervalRef.current = setInterval(() => {
        const elapsed = Date.now() - startTime;
        useRecordingStore.getState().setDuration(elapsed);
      }, 100);
    } else if (!isRecording) {
      if (intervalRef.current) clearInterval(intervalRef.current);
      useRecordingStore.getState().setDuration(0);
    }
    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
  }, [isRecording, isPaused, duration]);

  // Fade in/out transitions (UI-SPEC §7.4)
  useEffect(() => {
    opacity.value = withTiming(isRecording ? 1 : 0, { duration: 200 });
  }, [isRecording, opacity]);

  const animatedStyle = useAnimatedStyle(() => ({
    opacity: opacity.value,
  }));

  if (!isRecording) return null;

  return (
    <Animated.View style={animatedStyle} className="items-center">
      {/* Recording label per UI-SPEC §7.4 */}
      <View className="flex-row items-center gap-1.5 mb-1">
        <View className="w-2 h-2 rounded-full bg-primary" />
        <Text className="text-xs font-bold text-primary">
          {isPaused ? 'Call in progress — recording paused' : 'Recording'}
        </Text>
      </View>
      <Text className="font-sans text-4xl font-bold text-foreground text-center tracking-wider">
        {isPaused ? formatTime(duration) : formatTime(duration)}
      </Text>
    </Animated.View>
  );
}
```

Key design:
- Timer uses `setInterval` (not Reanimated) for the tick — simpler, no worklet overhead for state updates
- Duration stored in Zustand so all components access the same value
- Fade in/out via Reanimated `withTiming` (200ms per UI-SPEC §7.4)
- "Call in progress — recording paused" label shown when `isPaused` (UI-SPEC §10.1)
- Red dot + "REC ● Recording" label per UI-SPEC §7.4
