A) Create `src/hooks/useAudioCapture.ts`:
The primary orchestration hook that bridges AudioCaptureService → recordingStore → components. Also manages the amplitude buffer (SharedValue<number[]>) for the WaveformCanvas.

```typescript
import { useCallback, useRef } from 'react';
import { useSharedValue } from 'react-native-reanimated';
import { audioCaptureService } from '@/services/audioCaptureService';
import { useRecordingStore } from '@/stores/recordingStore';
import * as Haptics from 'expo-haptics';

const BUFFER_SIZE = 120; // ~2s at 60fps — matches WaveformCanvas expectations

export function useAudioCapture() {
  const {
    isRecording, isPaused, isProcessing, status,
    setRecording, setPaused, setProcessing, setError,
    setDuration, setMetering, setStatus, reset,
  } = useRecordingStore();

  // Amplitude buffer shared with Skia WaveformCanvas via Reanimated SharedValue
  const amplitudes = useSharedValue<number[]>(new Array(BUFFER_SIZE).fill(0));
  const durationRef = useRef(0);
  const smoothedAmplitude = useRef(0);

  const startRecording = useCallback(async () => {
    // Step 1: Start recording via service
    // (Haptic + spring fire in RecorderButton BEFORE this is called — per Research Pitfall 4)
    const success = await audioCaptureService.startRecording();
    if (!success) {
      setRecording(false);
      setError('Recording failed');
      Haptics.notificationAsync(Haptics.NotificationFeedbackType.Warning); // D-05
      return;
    }

    // Step 2: Update store to recording state
    setRecording(true);

    // Step 3: Wire metering callback from service → store + amplitude buffer
    // Exponential moving average smoothing (alpha 0.3) — Research §Waveform Canvas
    audioCaptureService.onMetering((value: number) => {
      const alpha = 0.3;
      const smoothed = smoothedAmplitude.current + alpha * (value - smoothedAmplitude.current);
      smoothedAmplitude.current = smoothed;

      // Update store metering for any component that reads it
      setMetering(smoothed);

      // Shift buffer: drop oldest, append new smoothed value
      const buf = [...amplitudes.value];
      buf.shift();
      buf.push(smoothed);
      amplitudes.value = buf;
    });

    // Step 4: Start duration timer
    durationRef.current = Date.now();
  }, [setRecording, setError, setMetering, amplitudes]);

  const stopRecording = useCallback(async (): Promise<string | undefined> => {
    // Step 1: Stop recording via service
    const uri = await audioCaptureService.stopRecording();

    // Step 2: Update duration
    if (durationRef.current) {
      const elapsed = Date.now() - durationRef.current;
      setDuration(elapsed);
    }

    if (!uri) {
      setError('Recording failed');
      Haptics.notificationAsync(Haptics.NotificationFeedbackType.Warning); // D-05
      return undefined;
    }

    // Step 3: Set processing state — transcription stub takes over
    setRecording(false);
    setProcessing(true); // Triggers ProcessingState UI

    return uri; // Return URI so useTranscription can process it
  }, [setRecording, setProcessing, setError, setDuration]);

  const retry = useCallback(() => {
    reset();
  }, [reset]);

  return {
    isRecording,
    isPaused,
    isProcessing,
    status,
    amplitudes,        // SharedValue<number[]> — consumed by WaveformCanvas
    startRecording,
    stopRecording,
    retry,
  };
}
```

Key design:
- `amplitudes: SharedValue<number[]>` returned from hook — consumed by WaveformCanvas via `useDerivedValue`
- Exponential moving average smoothing (alpha 0.3) applied before pushing to amplitude buffer (UI-SPEC §7.3)
- `startRecording` is the SECOND step (after haptic + spring in RecorderButton — consistent with Research Pitfall 4)
- `stopRecording` returns the URI so `useTranscription` can chain after it
- `retry()` resets the store to idle for error recovery
- Smoothed amplitude stored in a plain ref (useRef), not in state — avoids re-renders on every frame
- Buffer is a Reanimated SharedValue — Skia reads it on the UI thread via worklet

B) Create `src/hooks/useTranscription.ts`:
Orchestrates transcription stub, entries store update, and audio cleanup.

```typescript
import { useCallback } from 'react';
import { stubTranscription } from '@/services/transcriptionStub';
import { useEntriesStore } from '@/stores/entriesStore';
import { useRecordingStore } from '@/stores/recordingStore';
import { audioCaptureService } from '@/services/audioCaptureService';

export function useTranscription() {
  const { setProcessing, setError, setStatus } = useRecordingStore();
  const addEntry = useEntriesStore((s) => s.addEntry);

  const processRecording = useCallback(async (audioUri: string) => {
    try {
      // Step 1: Run stub transcription (1-3s simulated delay + file cleanup)
      const text = await stubTranscription(audioUri);

      // Step 2: Add entry to in-memory store (D-13 — Realm in Phase 2)
      addEntry({
        text,
        category: 'note', // Hardcoded for Phase 1 — Phase 3 does ML classification
        createdAt: new Date().toISOString(),
      });

      // Step 3: Audio file was cleaned up inside stubTranscription (VOIC-07)
      // Double-check cleanup for any remaining temp files
      try {
        const tempPath = audioCaptureService.getTempFilePath();
        if (tempPath && tempPath !== audioUri) {
          await audioCaptureService.cleanupTempFile(tempPath);
        }
      } catch { /* best-effort */ }

      // Do NOT call setProcessing(false) here — TranscriptionResult handles it
      // after 4s auto-dismiss, so the result card stays visible
    } catch (err) {
      setError('Transcription failed');
      setProcessing(false); // Reset on error — TranscriptionResult won't show
    }
  }, [addEntry, setProcessing, setError]);

  return { processRecording };
}
```

Key design:
- `processRecording` called by HomeScreen after `stopRecording()` returns a URI
- Returns `void` — component reads transcription text from entriesStore
- Entry added as category 'note' (hardcoded Phase 1, Phase 3 classifies)
- `setProcessing(false)` NOT called on success — TranscriptionResult handles it after 4s auto-dismiss
- Error isolation: if stub fails, error state + processing reset

**B) `src/components/ui/TranscriptionResult.tsx`** (UI-SPEC §7.6):
```typescript
import { useEffect, useState } from 'react';
import { View, Text } from 'react-native';
import Animated, {
  useSharedValue,
  useAnimatedStyle,
  withSpring,
  withTiming,
} from 'react-native-reanimated';
import { useEntriesStore } from '@/stores/entriesStore';
import { useRecordingStore } from '@/stores/recordingStore';

interface TranscriptionResultProps {
  visible: boolean;
}

export function TranscriptionResult({ visible }: TranscriptionResultProps) {
  const entry = useEntriesStore((state) => state.entries[0]); // latest entry
  const setProcessing = useRecordingStore((s) => s.setProcessing);
  const [show, setShow] = useState(false);
  const opacity = useSharedValue(0);
  const translateY = useSharedValue(20);

  // Show latest transcription result for 4s, then auto-dismiss (UI-SPEC §7.6)
  useEffect(() => {
    if (visible && entry) {
      setShow(true);
      opacity.value = withSpring(1);
      translateY.value = withSpring(0);

      const timer = setTimeout(() => {
        opacity.value = withTiming(0, { duration: 300 });
        translateY.value = withTiming(20, { duration: 300 });
        // After exit animation, hide component and return to idle
        setTimeout(() => {
          setShow(false);
          setProcessing(false); // Return to idle state — completes full pipeline
        }, 350);
      }, 4000); // 4s display (UI-SPEC §7.6)

      return () => clearTimeout(timer);
    } else {
      setShow(false);
    }
  }, [visible, entry, opacity, translateY, setProcessing]);

  const animatedStyle = useAnimatedStyle(() => ({
    opacity: opacity.value,
    transform: [{ translateY: translateY.value }],
  }));

  if (!show || !entry) return null;

  return (
    <Animated.View style={animatedStyle} className="mx-8">
      <View className="bg-card rounded-2xl border-2 border-border/30 px-5 py-4 shadow-[2px_2px_0px_theme(colors.border)] gap-3">
        {/* Transcription body text (UI-SPEC §7.6) */}
        <Text className="font-sans text-base font-medium text-foreground leading-6">
          {entry.text}
        </Text>

        {/* Category badge — hardcoded 'note' for Phase 1 (UI-SPEC §10.1) */}
        <View className="flex-row items-center justify-between">
          <View className="bg-accent px-2 py-0.5 rounded-full">
            <Text className="text-xs font-bold text-accent-foreground">
              Note
            </Text>
          </View>
        </View>

        {/* Recovery prompt (UI-SPEC §7.6) */}
        <Text className="font-sans text-xs font-medium text-primary text-center mt-1">
          Tap to record another thought
        </Text>
      </View>
    </Animated.View>
  );
}
```

interface ErrorBannerProps {

export function ErrorBanner({ visible, onRetry }: ErrorBannerProps) {
